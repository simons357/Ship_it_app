\paragraph{Definitions.}
$E=\|v\|_2^2$,\;
$X=\|A^{1/2}v\|_2^2$,\;
$Y=\|Av\|_2^2$,\;
$Z=\|A^{3/2}v\|_2^2$,\;
$\Lambda=Y/X$,\;
$D_s=Z-\Lambda Y$,\;
$T_c=-\langle B(v,v),A(A-\Lambda)v\rangle$.

\paragraph{Lemma$\star$ (shape).}
$(T_c)_+^2\le C_{\mathrm{geom}}\,D_s\,E\,Y$, or
$\mathcal{R}_\star=(T_c)_+^2/(D_s E Y)$ uniformly bounded.

\paragraph{Open estimate.}
Product-class control of $|T_c|$ from Leray-scale quantities (HH-channel bottleneck in the Bony diagnostic).

\paragraph{$\Phi$-renorm.}
$\Gamma=ru_\theta$, $\Phi=\Gamma/r^2$; identity $r^{-4}\partial_z(\Gamma^2)=\partial_z(\Phi^2)$;
dissipation label $\dot H^{1.3}$; barrier $\|u^r/r\|_\infty$ open.
