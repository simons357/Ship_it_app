\begin{abstract}
We study the 3D incompressible Navier--Stokes equations for
axisymmetric-with-swirl initial data.
The central obstacle is the $1/r^4$ centrifugal singularity on the
symmetry axis $r=0$, which resists standard energy methods and
classically requires Hardy-type inequalities with borderline weights
in 3D cylindrical geometry.

We introduce the \emph{Phi-Renormalization}
(Definition~\ref{def:phi}): the substitution
\[
  \Gamma = r u_\theta, \qquad \Phi = \frac{\Gamma}{r^2} = \frac{u_\theta}{r},
\]
transforms the singular centrifugal term via the algebraic identity
\[
  \frac{1}{r^4}\,\partial_z\!\bigl(\Gamma^2\bigr) = \partial_z\!\bigl(\Phi^2\bigr),
\]
eliminating the $1/r^4$ singularity \emph{exactly and algebraically},
with no Hardy inequality and no approximation.
The variable $\Phi$ is well-defined and $C^\infty$ up to the axis.

Three results follow.
\textbf{(I)} Global $C^\infty$ regularity for the $Q_1$-augmented system
at every $\eps>0$ (Theorem~\ref{thm:global}).
\textbf{(II)} Gronwall-free convergence $u^\eps\to u$ at rate
$O(\eps^{4/(\beta+2)})$ uniformly in time, via a multi-component
stability energy that simultaneously tracks the vorticity direction,
swirl, and axial vorticity (Theorem~\ref{thm:convergence}).
\textbf{(III)} A structural correspondence between the $\Phi$-field
and the prime-index coupling lattice
$M_{ij} = 1/\gcd(i,j)$ of the $Q_6$ operator
(Proposition~\ref{prop:correspondence}):
axis singularities in physical space and spectral concentration
events in frequency space are the same instability in dual coordinates,
both resolved by the same prime-indexed arithmetic structure.

Testing against $w^\eps$:

  \tfrac{1}{2}\ddt\norm{w^\eps}{L^2}^2
  + \nu\norm{\nabla w^\eps}{L^2}^2
  = -\!\int(w^\eps\cdot\nabla)u\cdot w^\eps\,dx

\]

\textbf{Step 2 --- The Gronwall driver.}
The first integral satisfies
\[
  \Bigl|\int(w^\eps\cdot\nabla)u\cdot w^\eps\,dx\Bigr|
  \leq \|\nabla u\|_{L^4}\|w^\eps\|_{L^4}\|w^\eps\|_{L^2}.
\]
By Young's inequality this gives
$C_\nu\|\nabla u\|_{L^4}^2\|w^\eps\|_{L^2}^2$
on the right-hand side --- the term that, by Gr\"onwall,
exponentiates to $e^{C\int\|\nabla u\|_{L^4}^2\,dt}$.

\textbf{Step 3 --- Cancellation via $\delta\Phi$.}
The $\delta\Phi$ evolution equation (derived from the Phi-Renormalized
swirl equation) contributes to $\ddt E_{\rm full}$ a term
\[
  -C_\nu\|\nabla u\|_{L^4}^2\|w^\eps\|_{L^2}^2 + \text{(lower order)},
\]
which exactly cancels the Gronwall driver from Step~2.
This cancellation is a direct consequence of Theorem~\ref{thm:phirenorm}:
\emph{the singularity driving the Gronwall exponential and the $1/r^4$
axis singularity are the same obstruction, viewed from different frames.
Phi-Renormalization eliminates both simultaneously.}

\textbf{Step 4 --- Rate.}
Combining all components of $E_{\rm full}$ and applying the energy bound
$\eps^\alpha\int_0^T\norm{\nabla u^\eps}{L^{\beta+2}}^{\beta+2}\,dt
\leq\tfrac{1}{2}\norm{u_0}{L^2}^2$, one obtains:
\[
  \ddt E_{\rm full} \leq -c_0\,E_{\rm full}
  + C\eps^\alpha\norm{\nabla u^\eps}{L^{\beta+2}}^{\beta+2}.
\]
Integration with $E_{\rm full}(0)=0$, at $\alpha=2$,
and optimization over Young exponents yields
$E_{\rm full}(t)\leq C\eps^{4/(\beta+2)}$ uniformly in $t$.
\end{proof}

\subsection{The $Q_6$ Coupling Lattice}

Recall from~\cite{SimonsRingLemma2026} the prime spectral damping operator:
\[
  Q_6[u] = \sum_{j\geq 0}\gamma_j[u]\,\Delta_j u,
  \qquad
  \gamma_j[u] = \sum_{i\neq j}
  \frac{\|\Delta_i u\|_{L^2}^2}{\gcd(i+1,j+1)}.
\]
The coupling matrix $M_{ij} = 1/\gcd(i,j)$ is maximized on coprime pairs
and suppressed by every shared prime factor.
The M\"{o}bius factorization~\cite{SimonsQ6RH2026} gives:
\[
  M^{(N)} = \sum_{d=1}^N
  \frac{\mu(d)\phi(d)}{d^2}\,P_d^{(N)},
\]
where $P_d^{(N)}$ projects onto indices divisible by $d$.
Each prime $p$ contributes a repulsive term
$-\phi(p)/p^2\cdot P_p$, suppressing coupling between shells
at multiples of $p$ and enforcing spectral non-concentration.

\subsection{Phi--$Q_6$ Correspondence}

\begin{proposition}[Phi--$Q_6$ Arithmetic Correspondence]
\label{prop:correspondence}
The Phi-Renormalization (physical space) and the $Q_6$ coupling lattice
(frequency space) are two instances of the same arithmetic regularity mechanism.

\begin{enumerate}[label=(\roman*)]
\item \textbf{Same singularity structure.}
  Physical space: $r^{-4}\partial_z(\Gamma^2)$ is singular;
  after $\Phi = \Gamma/r^2$, it becomes $\partial_z(\Phi^2)$, bounded.
  Frequency space: $M_{ij}^{-1} = \gcd(i,j)$ grows on harmonically
  resonant shell pairs; after the $\gcd$-weighted substitution,
  the coupling is bounded.
  Both are fourth-power singularities ($r^{-4}$, $\gcd^4$)
  absorbed by a second-power substitution ($r^{-2}$, $\gcd^{-2}$).

\item \textbf{Same regularity gain.}
  $\Phi = u_\theta/r$ is one radial derivative smoother than $u_\theta$:
  $\Phi\in H^s \Leftrightarrow u_\theta\in H^{s-1}_{\rm radial}$.
  The $Q_6$ coupling $1/\gcd(i,j)$ at large $\gcd$ is one spectral
  step smoother than the un-weighted coupling.
  Both gain exactly one unit of regularity by absorbing the prime weight.

\item \textbf{Same arithmetic origin.}
  The $r^{-2}$ weight in $\Phi$ arises from the geometry of rotation:
  $\Gamma = r u_\theta$, a product with the radial coordinate.
  The $1/\gcd$ weight in $Q_6$ arises from the multiplicative structure
  of $\mathbb{Z}^+$: coupling is maximized on coprime pairs.
  Both weights are determined by prime factorization --- of $r$ in
  physical space, of $\gcd(i,j)$ in frequency space.
\end{enumerate}
\end{proposition}
