\begin{abstract}
Let $Q_N[i,j]=1/(\gcd(i,j)\sqrt{ij})$ and
$v_{\mathrm{alt}}[k]=(-1)^{k+1}/\sqrt{k}$.
Earlier Route~C drafts claimed $R(v_{\mathrm{alt}})\to -1/(2\pi)$ via a
M\"obius identity that fails off-diagonal.
We kill that path and reframe the architecture: numerics support
$\lambda_{\min}(Q_N)/\log N\to -1/(2\pi)$ while $R(v_{\mathrm{alt}})$
diverges from the target.
We prove a parity split of the alternating quadratic form and record two
honest analytic gaps. No RH claim.
\end{abstract}

\section{Operator and trial vectors}

\[
Q_N[i,j]=\frac{1}{\gcd(i,j)\sqrt{ij}},\qquad
v_{\mathrm{alt}}[k]=\frac{(-1)^{k+1}}{\sqrt{k}}.
\]
Then
\[
v_{\mathrm{alt}}^\top Q_N v_{\mathrm{alt}}
=\sum_{i,j=1}^N\frac{(-1)^{i+j}}{\gcd(i,j)\,ij}.
\]
Write $\hat v_{\mathrm{alt}}=v_{\mathrm{alt}}/\|v_{\mathrm{alt}}\|$ and
$R(w)=w^\top Q_N w/\|w\|_2^2$.

\section{What fails (killed)}

\begin{proposition}[Archived Lemma~A is false]\label{prop:lemmaAfalse}
The decomposition
$1/\gcd(i,j)=\sum_{d\mid\gcd(i,j)}\mu(d)\varphi(d)/d^2$
fails already at $\gcd=2$ ($1/2$ vs.\ $3/4$).
The induced matrix expansion is not $Q_N$.
\end{proposition}

\begin{proposition}[Gap~A for $v_{\mathrm{alt}}$ is false as stated]\label{prop:valtfalse}
For $N=500,1000$, numerics give $R(v_{\mathrm{alt}})\approx -0.64,-0.73$
while $-1/(2\pi)\approx -0.159$; the ratio grows with $N$.
Similarly $v_{\mathrm{alt}}^\top Q_N v_{\mathrm{alt}}$ is $\sim 5\times$
more negative than $-\log N/(2\pi)$ and diverges in ratio.
Certificate: \texttt{scripts/route\_c\_gap\_a\_verify.py}.
\end{proposition}

\begin{remark}
High overlap $|\langle v_N^*,\hat v_{\mathrm{alt}}\rangle|\approx 0.95$
does \emph{not} force $R(v_{\mathrm{alt}})\approx\lambda_{\min}$:
$\lambda_{\min}<R(v_{\mathrm{alt}})$ always in checks.
\end{remark}

\section{Proved: parity split}

\begin{lemma}[Parity split]\label{lem:parity}
\[
\sum_{i,j=1}^N\frac{(-1)^{i+j}}{\gcd(i,j)\,ij}
=\sum_{\substack{d\le N\\ d\text{ odd}}}\frac{T_-(N/d)}{d^3}
+\sum_{\substack{d\le N\\ d\text{ even}}}\frac{T_+(N/d)}{d^3},
\]
where
\[
T_-(M)=\sum_{\gcd(m,n)=1}\frac{(-1)^{m+n}}{mn},\qquad
T_+(M)=\sum_{\gcd(m,n)=1}\frac1{mn}.
\]
\end{lemma}

\begin{proof}
Write $i=dm$, $j=dn$ with $\gcd(m,n)=1$. The contribution is
$(-1)^{d(m+n)}/(d^3 mn)$, which equals $(-1)^{m+n}/(d^3 mn)$ when $d$ is odd
and $1/(d^3 mn)$ when $d$ is even.
\end{proof}

\begin{lemma}[Coprime M\"obius forms]\label{lem:mobiusT}
\[
T_-(M)=\sum_{d=1}^M\frac{\mu(d)}{d^2}
\Bigl(\sum_{m\le M/d}\frac{(-1)^{dm}}{m}\Bigr)^{\!2},\quad
T_+(M)=\sum_{d=1}^M\frac{\mu(d)}{d^2}
\Bigl(\sum_{m\le M/d}\frac1m\Bigr)^{\!2}.
\]
\end{lemma}

\begin{proof}
Expand $\sum_{\gcd(m,n)=1} f(m)f(n)/(mn)$ via M\"obius inversion
$\mathbf 1_{\gcd=1}=\sum_{d\mid(m,n)}\mu(d)$ and change variables.
\end{proof}

\begin{remark}
Even-$d$ blocks contribute \emph{positively} and cancel a large fraction of
odd-$d$ negativity (e.g.\ at $N=1000$, even $\approx +4.7$, odd $\approx -10.1$).
This refutes archived heuristics that even blocks alone drive $-1/(2\pi)$.
\end{remark}

\section{Architecture (conditional, reframed)}

\begin{theorem}[Route~C template, conditional]
Assume Gap~A$'$ (spectral limit
$\lambda_{\min}(Q_N)/\log N\to -1/(2\pi)$ with a rigorous trial-vector lower bound)
and Gap~B (uniform $\lambda_2(Q_N)-\lambda_{\min}(Q_N)\ge c_0>0$).
Then Davis--Kahan perturbation yields
$\|v_N^*-\hat v_{\mathrm{alt}}\|\ll N^{-\eta}$ for some $\eta>0$.
\end{theorem}

\begin{remark}
The old Gap~A target $R(v_{\mathrm{alt}})\to -1/(2\pi)$ must not be used:
Proposition~\ref{prop:valtfalse}. A bridge to zero-density / RH criteria remains
separate even if Gaps~A$'$--B close.
\end{remark}

\begin{gap}
\begin{enumerate}[leftmargin=1.2em]
\item \textbf{Gap~A$'$ (spectral).}
Prove $\displaystyle\lim_{N\to\infty}\frac{\lambda_{\min}(Q_N)}{\log N}=-\frac{1}{2\pi}$
via a trial vector with provable Rayleigh $\sim -(\log N)/(2\pi)$,
\emph{not} $v_{\mathrm{alt}}$ as stated in archived drafts.
\item \textbf{Gap~B (uniform gap).}
Prove $\lambda_2(Q_N)-\lambda_{\min}(Q_N)\ge c_0>0$ uniformly in $N$
(numeric: $\approx 0.09$--$0.14$ and slowly increasing through $N=5000$).
\end{enumerate}
\end{gap}

\section{Partial progress on growth (optional)}

Using Lemmas~\ref{lem:parity}--\ref{lem:mobiusT}, one can aim for
\[
v_{\mathrm{alt}}^\top Q_N v_{\mathrm{alt}}=-c_1(\log N)^2+O(\log N)
\]
with explicit $c_1>0$ from even/odd M\"obius blocks.
This closes the \emph{growth order} but not the $-1/(2\pi)$ constant for
$v_{\mathrm{alt}}$.

\section{What is not claimed}

\begin{itemize}[leftmargin=1.2em]
\item RH or unconditional zero-density theorems.
\item $R(v_{\mathrm{alt}})\to -1/(2\pi)$ or archived Lemma~A.
\item Full-spectrum Bridge floors $\lambda_{\min}(Q_N)>-1/2$ (false).
\end{itemize}
